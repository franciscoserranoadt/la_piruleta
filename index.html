  <script>
    const productos = [
      { id: 1, nombre: "Flat White Doble", precio: 2.80, cat: "Café", emoji: "☕" },
      { id: 2, nombre: "Cappuccino Avena", precio: 2.90, cat: "Café", emoji: "🥛" },
      { id: 3, nombre: "Cold Brew Nitro", precio: 3.40, cat: "Café", emoji: "🧊" },
      { id: 4, nombre: "Hogaza Masa Madre 1kg", precio: 4.20, cat: "Pan", emoji: "🥖" },
      { id: 5, nombre: "Pan Centeno 500g", precio: 3.10, cat: "Pan", emoji: "🍞" },
      { id: 6, nombre: "Croissant Mantequilla", precio: 1.95, cat: "Bollería", emoji: "🥐" },
      { id: 7, nombre: "Pain au Chocolat", precio: 2.30, cat: "Bollería", emoji: "🍫" },
      { id: 8, nombre: "Tostada Ibérica AOVE", precio: 4.50, cat: "Tostadas", emoji: "🥪" }
    ];

    let carrito = {};
    let categoriaActiva = "Todos";

    function filtrarCategoria(cat) {
      categoriaActiva = cat;
      ["Todos", "Café", "Pan", "Bollería", "Tostadas"].forEach(c => {
        const btn = document.getElementById("cat-" + c);
        if (btn) {
          if (c === cat) {
            btn.className = "px-3.5 py-1.5 rounded-full bg-stone-900 text-white font-semibold whitespace-nowrap shadow-sm";
          } else {
            btn.className = "px-3.5 py-1.5 rounded-full bg-white text-stone-600 border border-stone-200 whitespace-nowrap shadow-sm";
          }
        }
      });
      renderCatalogo();
    }

    function renderCatalogo() {
      const contenedor = document.getElementById("catalogo");
      const prodsFiltrados = categoriaActiva === "Todos" 
        ? productos 
        : productos.filter(p => p.cat === categoriaActiva);

      contenedor.innerHTML = prodsFiltrados.map(p => {
        const cant = carrito[p.id] || 0;
        return `
          <div class="bg-white p-3.5 rounded-2xl border ${cant > 0 ? 'border-[#c85a32]/60 ring-2 ring-[#c85a32]/20' : 'border-stone-200/80'} shadow-sm flex flex-col justify-between transition active:scale-[0.99]">
            <div class="flex items-start justify-between">
              <span class="text-2xl">${p.emoji}</span>
              ${cant > 0 ? `<span class="bg-[#c85a32] text-white text-xs font-bold px-2 py-0.5 rounded-full font-mono">${cant}</span>` : ''}
            </div>
            <div class="my-2">
              <h3 class="font-bold text-stone-900 text-sm leading-snug">${p.nombre}</h3>
              <p class="text-xs text-stone-400 font-medium">${p.cat}</p>
            </div>
            <div class="flex items-center justify-between pt-1">
              <span class="font-brand font-bold text-stone-900 text-sm">${p.precio.toFixed(2)} €</span>
              <div class="flex items-center gap-1">
                ${cant > 0 ? `
                  <button onclick="modificarCantidad(${p.id}, -1)" class="w-7 h-7 rounded-lg bg-stone-100 border border-stone-200 text-stone-700 flex items-center justify-center font-bold text-sm active:bg-stone-200">-</button>
                ` : ''}
                <button onclick="modificarCantidad(${p.id}, 1)" class="w-7 h-7 rounded-lg bg-[#c85a32] text-white flex items-center justify-center font-bold text-sm shadow-sm active:scale-90">+</button>
              </div>
            </div>
          </div>
        `;
      }).join('');
      actualizarTotales();
    }

    function modificarCantidad(id, delta) {
      carrito[id] = (carrito[id] || 0) + delta;
      if (carrito[id] <= 0) delete carrito[id];
      renderCatalogo();
    }

    function calcularTotal() {
      let total = 0;
      let cantidad = 0;
      for (const id in carrito) {
        const prod = productos.find(p => p.id == id);
        if (prod) {
          total += prod.precio * carrito[id];
          cantidad += carrito[id];
        }
      }
      return { total, cantidad };
    }

    function actualizarTotales() {
      const { total, cantidad } = calcularTotal();
      document.getElementById("cantTotal").innerText = cantidad;
      document.getElementById("precioTotal").innerText = total.toFixed(2).replace('.', ',') + " €";
      document.getElementById("btnCobrar").disabled = cantidad === 0;
    }

    function abrirCobro() {
      const { total, cantidad } = calcularTotal();
      if (cantidad === 0) return;
      document.getElementById("totalModal").innerText = total.toFixed(2).replace('.', ',') + " €";
      document.getElementById("resumenModal").innerText = `${cantidad} productos • Ticket #${Math.floor(1000 + Math.random() * 9000)}`;

      const desglose = document.getElementById("desgloseItems");
      let html = "";
      for (const id in carrito) {
        const p = productos.find(item => item.id == id);
        const sub = p.precio * carrito[id];
        html += `<div class="flex justify-between text-stone-600"><span>${carrito[id]}x ${p.nombre}</span><span class="font-medium">${sub.toFixed(2)} €</span></div>`;
      }
      desglose.innerHTML = html;

      document.getElementById("seccionMetodos").classList.remove("hidden");
      document.getElementById("pantallaExito").classList.add("hidden");
      document.getElementById("modalCobro").classList.remove("hidden");
    }

    function cerrarCobro() {
      document.getElementById("modalCobro").classList.add("hidden");
    }

    function finalizarVenta(metodo) {
      document.getElementById("seccionMetodos").classList.add("hidden");
      document.getElementById("exitoTitulo").innerText = `¡Cobro con ${metodo} Registrado!`;
      document.getElementById("pantallaExito").classList.remove("hidden");
    }

    function reiniciarTicket() {
      carrito = {};
      renderCatalogo();
    }

    function reiniciarTodo() {
      carrito = {};
      renderCatalogo();
      cerrarCobro();
    }

    function compartirWhatsApp() {
      const { total } = calcularTotal();
      let detalle = "";
      for (const id in carrito) {
        const p = productos.find(item => item.id == id);
        detalle += `• ${carrito[id]}x ${p.nombre}%0A`;
      }
      const texto = `🍭 *LA PIRULETA - Ticket de Compra*%0A---------------------------%0A${detalle}---------------------------%0A*Total: ${total.toFixed(2)} €*%0A¡Muchas gracias por tu visita! 😊`;
      window.open(`https://wa.me/?text=${texto}`, '_blank');
    }

    renderCatalogo();
  </script>
